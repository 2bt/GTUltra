# The original codebase relies on lax conversion rules; keep -fpermissive for
# the C++ translation units so the incremental C -> C++ migration can proceed.
# (No effect on the remaining C sources in asm/; host tools are plain C.)
add_compile_options(-Wall)
# -fpermissive: allow the lax C-style conversions the code relies on.
# -Wno-narrowing: the code initialises char[] tables with values > 127 (byte
#   graphics data), which is a hard error in C++ braced-init but fine in C.
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fpermissive>)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wno-narrowing>)

# Embedded resources (players, icon, UI font) via CMake HEX → embed catalog
# ---------------------------------------------------------------------------
# embed_resources(<out_inc> <file>...) — HEX-encode files into a C++ include.
function(embed_resources out_file)
    if(NOT out_file)
        message(FATAL_ERROR "embed_resources() requires an output path")
    endif()
    if(${ARGC} LESS 2)
        message(
            FATAL_ERROR
            "embed_resources() requires at least one input file"
        )
    endif()

    set(content
        "// Auto-generated by embed_resources() — do not edit.\n#include <cstddef>\n#include <cstdint>\n\n"
    )

    foreach(path IN LISTS ARGN)
        if(NOT EXISTS "${path}")
            message(FATAL_ERROR "embed_resources: missing ${path}")
        endif()

        get_filename_component(fname "${path}" NAME)
        string(MAKE_C_IDENTIFIER "${fname}" sym)

        file(READ "${path}" hex HEX)
        string(REGEX REPLACE "([0-9a-fA-F][0-9a-fA-F])" "0x\\1," hex "${hex}")

        string(
            APPEND
            content
            "static constexpr uint8_t k_bytes_${sym}[] = { ${hex} };\n"
            "static constexpr std::size_t k_size_${sym} = sizeof(k_bytes_${sym});\n\n"
        )
    endforeach()

    # Avoid touching the timestamp when contents are unchanged.
    set(tmp "${out_file}.tmp")
    file(WRITE "${tmp}" "${content}")
    execute_process(
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${tmp}" "${out_file}"
    )
    file(REMOVE "${tmp}")
endfunction()
